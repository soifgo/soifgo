#include <ESP8266WiFi.h>
#include <PubSubClient.h>

// Wi-Fi settings
const char* ssid = "TPLINK";        // Wi-Fi SSID
const char* password = "abcd1234";  // Wi-Fi password

// MQTT settings
const char* mqtt_server = "test.mosquitto.org"; // Broker address (or direct IP: "85.119.83.194")
const int   mqtt_port   = 1883;                 // MQTT port
const char* mqtt_recv_topic = "soifgo_send";    // Fixed topic for receiving

WiFiClient espClient;
PubSubClient client(espClient);

void setup_wifi() {
  delay(10);
  Serial.println();
  Serial.print("Connecting to ");
  Serial.println(ssid);

  WiFi.begin(ssid, password);

  while (WiFi.status() != WL_CONNECTED) {
    delay(500);
    Serial.print(".");
  }

  Serial.println("");
  Serial.println("WiFi connected");
  Serial.print("IP address: ");
  Serial.println(WiFi.localIP());
  Serial.print("MAC address: ");
  Serial.println(WiFi.macAddress());
}

// Callback when a new MQTT message is received
void mqttCallback(char* topic, byte* payload, unsigned int length) {
  String message;
  for (unsigned int i = 0; i < length; i++) {
    message += (char)payload[i];
  }
  if (message.length() > 0) {
    // Only print messages from the fixed receive topic
    if (String(topic) == mqtt_recv_topic) {
      Serial.println(message);
    }
  }
}

void reconnect() {
  while (!client.connected()) {
    String clientId = "ESP8266Client-" + String(ESP.getChipId());
    if (client.connect(clientId.c_str())) {
      // Subscribe only to the fixed receive topic
      client.subscribe(mqtt_recv_topic);
    } else {
      delay(2000);
    }
  }
}

void setup() {
  Serial.begin(115200);
  setup_wifi();
  client.setServer(mqtt_server, mqtt_port);
  client.setCallback(mqttCallback);
}

void loop() {
  if (!client.connected()) {
    reconnect();
  }
  client.loop();

  // If data comes from Serial → split into topic and payload
  static String serialBuffer = "";
  while (Serial.available()) {
    char c = Serial.read();
    if (c == '\n') {
      serialBuffer.trim();
      if (serialBuffer.length() > 0) {
        int sepIndex = serialBuffer.indexOf(':');
        if (sepIndex > 0) {
          String topicPart   = serialBuffer.substring(0, sepIndex);   // e.g. temp1
          String payloadPart = serialBuffer.substring(sepIndex + 1); // e.g. 123
          String fullTopic   = "soifgo_" + topicPart;                // e.g. soifgo_temp1
          client.publish(fullTopic.c_str(), payloadPart.c_str());
        }
      }
      serialBuffer = ""; // Clear buffer after sending
    } else {
      serialBuffer += c;
    }
  }
}
