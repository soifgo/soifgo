  <script>
    function sendBluetooth(text) {
      if (window.soifgo) {
        window.soifgo.CallSub('html_bluetooth_rx', true, text);
      }
    }

    function html_bluetooth_tx(data) {
      document.getElementById('raw').innerText = 'Received: ' + data;

      if (data.indexOf('SENSOR:') === 0) {
        var val = data.replace('SENSOR:', '').trim();
        document.getElementById('value').innerText = val;
      }
    }
    window.html_bluetooth_tx = html_bluetooth_tx;
  </script>
